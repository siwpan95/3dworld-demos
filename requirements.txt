# 3dworld-demos —— 运行依赖（单一事实来源）
#
# 2026-10-22 变更（重要）
# ----------------------
# 此前 39 个 demo 的头部都有一句
#     sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))
# 用来把「相邻目录里的引擎源码」塞进搜索路径。演示仓从引擎仓剥离之后，这句指向的
# 已经不是引擎根（而是演示仓自己的父目录），于是所有 demo 都会
#     ModuleNotFoundError: No module named 'world3d'
#
# 现在一律由 **pip 提供引擎**：examples/*.py 里不再有任何 sys.path 引导，直接
#     from world3d.platform.app import DemoApp, run
# 完整背景、仓间关系与例外说明见 docs/依赖与版本.md。
#
# 版本号为什么用范围而不是钉死
# ----------------------------
# 演示不锁定引擎小版本（0.3.x 都接受）：它们是能力的「演示形态」，引擎发新版就跟着涨。
# 需要**逐位可复现**（截图对照 / 发版归档）时用旁边的 requirements.lock.txt。

# ① 引擎本体：发行名 3dworld，导入名 world3d（Python 不能以数字开头做包名）
#    传递依赖自动带进来：pyglet==2.1.16、PyYAML>=6、watchdog>=4
#
# ⚠️ 2026-10-26 上限从 <0.4 抬到 <0.5（下限抬到 0.4.1）—— 这个上限曾经把引擎
#    **锁死在 0.3.0**：pip 只能装到 0.3.0，而 0.3.0 的 wheel 里没有
#    world3d/physics、anim、engine/terrain_lod、engine/terrain_clipmap，
#    于是 rigidward / cdlod_terrain 这类演示全部 "引擎没发布这个模块"。
#    实测 0.4.1 的 wheel **全部带齐**（含 wardfall 要的
#    fusion_audio.ensure_wardfall_ost），引擎从来没欠过我们，是这条约束欠的。
# ⚠️ 2026-10-27 上限从 <0.5 抬到 <0.6（下限抬到 0.5.0）：0.5.0 的 wheel 带来了
#    world3d/engine/compute.py（GPU 计算调度核心）——computeward 的 GPU 主路径
#    依赖它；旧 0.4.1 环境下该 demo 自动降级 CPU 路径仍可跑，但抬下限后
#    「pip install -r requirements.txt」即得完整体验。
# ⚠️ 2026-10-27（二）上限从 <0.6 抬到 <0.7（下限抬到 0.6.0）：0.6.0 的 wheel
#    带来 world3d/logic/fov.py（recursive shadowcasting）——brogueward 的视野
#    与怪物惊醒走引擎件（旧 0.5.0 下自动降级内置同款 shadowcasting）。
# ⚠️ 2026-11-14 上限从 <0.7 抬到 <0.8（下限抬到 0.7.3）：引擎 0.7.3 发版带来
#    world3d/engine/spectrum.py + spectral_glsl.py（《晶极 / Crystalward》反哺的光谱
#    能力）；crystalward.py 按仓惯例自包含，不 import 它们，但旧上限会让新 clone
#    只能装到 0.6.x，与引擎现状脱节。0.7.3 已在 PyPI 实测可装。
3dworld>=0.7.3,<0.8

# ② 可选增强：examples/scene_walker.py 的马尔可夫演出变体（纯标准库实现）
#    不装的话 scene_walker 只是没有发光变体，其余功能照常（ImportError 已兜住）
#    —— 2026-10-22 起不再是环境变量 ZHUANGTAIJI_PATH + 目录探测那一套
zhuangtaiji>=1.1.0,<2

# ③ numpy：examples/rigidward.py 依赖 world3d.physics.jolt_lite，那个模块
#    import numpy —— 但 numpy **不在 3dworld 的 install_requires 里**
#    （2026-10-26 实测：干净 venv 里 pip install 3dworld 之后
#     `from world3d.physics.jolt_lite import World` 直接 ModuleNotFoundError）。
#    引擎 0.4.1 也没声明它（Requires-Dist 里只有 pyglet/PyYAML/watchdog），
#    所以由本仓显式兜住；引擎下次发版时应把 numpy 收进它自己的依赖。
numpy>=1.24

# ℹ️ examples/rigidward.py 需要含 world3d.physics 的版本 —— 0.4.1 起就带齐了，
#    下限 0.4.1 已经把它兜住。万一拿到旧版，它会给出明确提示后退出，
#    而不是丢一个裸 ModuleNotFoundError。
